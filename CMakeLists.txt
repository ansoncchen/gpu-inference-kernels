cmake_minimum_required(VERSION 3.22)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
project(gpu_inference_kernels CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "")
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
set(CMAKE_CUDA_FLAGS_RELEASE "-O3")
find_package(OpenMP REQUIRED)
find_package(CUDAToolkit REQUIRED)
include_directories(include)

add_library(q4 STATIC src/q4.cpp)
target_link_libraries(q4 PUBLIC OpenMP::OpenMP_CXX)
add_library(q4gpu STATIC src/gpu_kernels.cu)
target_link_libraries(q4gpu PUBLIC q4)

add_executable(test_gpu tests/test_gpu.cpp)
target_link_libraries(test_gpu q4gpu CUDA::cudart_static)
add_executable(bench_gpu bench/bench_gpu.cpp)
target_link_libraries(bench_gpu q4gpu CUDA::cudart_static CUDA::cublas ${CMAKE_DL_LIBS} pthread)

add_library(q4cpu STATIC src/cpu_kernels.cpp)
target_compile_options(q4cpu PRIVATE -mavx2 -mfma -mf16c)
target_link_libraries(q4cpu PUBLIC q4)
add_executable(test_cpu tests/test_cpu.cpp)
target_compile_options(test_cpu PRIVATE -mavx2 -mf16c)
target_link_libraries(test_cpu q4cpu)

find_library(OPENBLAS_LIB openblas)
find_path(OPENBLAS_INC cblas.h)
if(OPENBLAS_LIB AND OPENBLAS_INC)
  add_executable(bench_cpu bench/bench_cpu.cpp)
  target_include_directories(bench_cpu PRIVATE ${OPENBLAS_INC})
  target_link_libraries(bench_cpu q4cpu ${OPENBLAS_LIB})
endif()

# Optional llama.cpp/ggml baseline: -DLLAMA_DIR=<llama.cpp checkout with build/>
set(LLAMA_DIR "" CACHE PATH "llama.cpp checkout, built in <dir>/build")
if(LLAMA_DIR)
  add_executable(bench_ggml baselines/bench_ggml.cpp)
  target_include_directories(bench_ggml PRIVATE ${LLAMA_DIR}/ggml/include)
  target_link_directories(bench_ggml PRIVATE ${LLAMA_DIR}/build/bin)
  target_link_libraries(bench_ggml q4 ggml ggml-base ggml-cuda ${CMAKE_DL_LIBS} pthread)
endif()
